TASK 2 — Access-lock middleware
Add middleware applied to all /developer/* routes (dashboard, criteria, outreach, 
crm, analytics):

async function requireCurrentCheckIn(req, res, next) {
  const profile = await getDeveloperProfile(req.user.developerProfileId);
  if (!profile.checkInFeatureEnabled) return next(); // feature off = never blocks
  const lastCheckIn = await getLatestCheckIn(profile.id);
  const isOverdue = !lastCheckIn || lastCheckIn.nextDueAt < new Date();
  if (isOverdue) return res.status(423).json({ locked: true, reason: "check_in_required" });
  next();
}

# Post Property

POST https://api.hellodata.ai/property
Content-Type: application/json

Insert a new property in the database. For example a new development.

Reference: https://docs.hellodata.ai/api-reference/api-reference/property/post-property

let request = NSMutableURLRequest(url: NSURL(string: "https://api.hellodata.ai/property")! as URL,

    let httpResponse = response as? HTTPURLResponse
    print(httpResponse)
  }
})

dataTask.resume()

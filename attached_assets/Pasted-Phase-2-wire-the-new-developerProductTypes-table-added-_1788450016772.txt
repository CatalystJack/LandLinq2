TASK 1 — Rewrite the classification engine
In server/developerClassificationService.ts, replace the existing 
classifyDealForProfile logic with:

export function classifyDealForProfile(deal, profile, productTypes): { 
  classification: 'passed' | 'review'; 
  matchedProductTypes: string[] 
} {
  const countyMatch = (profile.targetCounties || []).includes(deal.county) 
    || (profile.targetStates || []).includes(deal.state);

  if (!countyMatch) {
    return { classification: 'passed', matchedProductTypes: [] };
  }

  const matched: string[] = [];
  for (const pt of productTypes.filter(p => p.isActive)) {
    const acreagePass = parseFloat(deal.sizeAcres) >= parseFloat(pt.minAcres)
      && (!pt.maxAcres || parseFloat(deal.sizeAcres) <= parseFloat(pt.maxAcres));

    let rentPass: boolean;
    if (profile.rentMetric === 'psf') {
      rentPass = parseFloat(deal.topRentPSF) >= parseFloat(pt.minRentPsf);
    } else {
      rentPass = parseFloat(deal.avgRentPerUnit) >= parseFloat(pt.minRentPerUnit);
    }

    // QCT/DDA/OZ override still applies per product type: only rescues a
    // rent-only failure — acreage must still pass on its own.
    if (acreagePass && !rentPass) {
      const qualifiesForOverride =
        (deal.isQct && profile.qctOverridesRentMinimum) ||
        (deal.isDda && profile.ddaOverridesRentMinimum) ||
        (deal.isOz && profile.ozOverridesRentMinimum);
      if (qualifiesForOverride) rentPass = true;
    }

    if (acreagePass && rentPass) matched.push(pt.name);
  }

  return {
    classification: matched.length > 0 ? 'review' : 'passed',
    matchedProductTypes: matched,
  };
}

Update every call site of classifyDealForProfile (auto-send, bulk import, manual 
send, and any email-intake routing) to first fetch that profile's active 
developerProductTypes, pass them in, and store the returned matchedProductTypes 
array on the partnerDeveloperSends record alongside the classification.
